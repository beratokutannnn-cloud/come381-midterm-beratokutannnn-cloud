-- Q1: Fiyatı 50 TL'den yüksek ürünleri pahalıdan ucuza sırala.
-- Ne işe yarar: Kafenin en pahalı ürünlerini görüp fiyat politikasını gözden geçirmeyi sağlar.
SELECT name, price
FROM product
WHERE price > 50
ORDER BY price DESC;

--------------+-------

 Cheesecake   | 85.00
 Tost         | 70.00
 Latte        | 60.00

-- Q5: Her müşterinin kaç sipariş verdiğini müşteri adıyla listele.
-- Ne işe yarar: En sadık müşterileri ve hiç sipariş vermeyenleri ortaya çıkarır.
SELECT c.full_name, COUNT(o.order_id) AS siparis_sayisi
FROM customer c
LEFT JOIN orders o ON o.customer_id = c.customer_id
GROUP BY c.customer_id, c.full_name
ORDER BY siparis_sayisi DESC;

-- Q6: Her ürün kategorisinin ortalama fiyatını bul.
-- Ne işe yarar: Hangi kategorinin ortalama olarak daha pahalı olduğunu karşılaştırmayı sağlar.
SELECT category, ROUND(AVG(price)::numeric, 2) AS ortalama_fiyat
FROM product
GROUP BY category;
